package com.sean8.jeminai.core.accessibility.llm

import android.util.Log
import com.squareup.moshi.JsonAdapter
import com.squareup.moshi.Moshi
import com.squareup.moshi.Types
import com.squareup.moshi.kotlin.reflect.KotlinJsonAdapterFactory
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import okhttp3.MediaType.Companion.toMediaTypeOrNull
import okhttp3.MultipartBody
import okhttp3.OkHttpClient
import okhttp3.Request
import okhttp3.RequestBody
import java.io.IOException
import javax.inject.Inject
import javax.inject.Singleton

@Singleton
class ChatGPTService @Inject constructor() : LLMService {

    private val TAG = "ChatGPTService"

    private val client = OkHttpClient()

    // TODO: Store API key securely
    private val apiKey = "YOUR_OPENAI_API_KEY"

    private val moshi: Moshi = Moshi.Builder()
        .add(KotlinJsonAdapterFactory())
        .build()

    private val messages = mutableListOf<Message>()

    override suspend fun sendMessage(message: String, imageByteArray: ByteArray?): LLMResponse {
        messages.add(Message(role = "user", content = message))

        val messagesType = Types.newParameterizedType(List::class.java, Message::class.java)
        val messagesAdapter: JsonAdapter<List<Message>> = moshi.adapter(messagesType)
        val messagesJson = messagesAdapter.toJson(messages)

        val requestBodyBuilder = MultipartBody.Builder().setType(MultipartBody.FORM)
            .addFormDataPart("model", "gpt-4-vision")
            .addFormDataPart("messages", messagesJson)

        if (imageByteArray != null) {
            requestBodyBuilder.addFormDataPart(
                "file",
                "screenshot.png",
                RequestBody.create("image/png".toMediaTypeOrNull(), imageByteArray)
            )
        }

        val requestBody = requestBodyBuilder.build()

        val request = Request.Builder()
            .url("https://api.openai.com/v1/chat/completions")
            .header("Authorization", "Bearer $apiKey")
            .post(requestBody)
            .build()

        return withContext(Dispatchers.IO) {
            try {
                client.newCall(request).execute().use { response ->
                    if (!response.isSuccessful) {
                        Log.e(TAG, "Unexpected code $response")
                        LLMResponse(null, null)
                    } else {
                        val responseBody = response.body?.string()
                        val chatGPTResponseAdapter = moshi.adapter(ChatGPTResponse::class.java)
                        val chatGPTResponse = chatGPTResponseAdapter.fromJson(responseBody!!)
                        val assistantMessage = chatGPTResponse?.choices?.firstOrNull()?.message?.content

                        // Add assistant's response to the conversation
                        messages.add(Message(role = "assistant", content = assistantMessage ?: ""))

                        return@withContext parseAssistantMessage(assistantMessage)
                    }
                }
            } catch (e: IOException) {
                Log.e(TAG, "Error: ${e.message}", e)
                LLMResponse(null, null)
            }
        }
    }

    override fun resetSession() {
        messages.clear()
    }

    private fun parseAssistantMessage(messageContent: String?): LLMResponse {
        if (messageContent == null) {
            return LLMResponse(null, null)
        }

        return try {
            val adapter = moshi.adapter(LLMResponse::class.java)
            adapter.fromJson(messageContent) ?: LLMResponse(null, null)
        } catch (e: Exception) {
            Log.e(TAG, "Failed to parse assistant message: ${e.message}", e)
            LLMResponse(null, null)
        }
    }
}

data class Message(
    val role: String,
    val content: String
)

data class ChatGPTResponse(
    val id: String,
    val choices: List<Choice>
)

data class Choice(
    val index: Int,
    val message: Message
)

package com.sean8.jeminai.core.accessibility.llm

import android.app.Notification

import kotlinx.coroutines.launch
import java.io.ByteArrayOutputStream
import javax.inject.Inject

@AndroidEntryPoint
class ServerMockService : Service() {

    @Inject
    lateinit var screenshotExtractor: ProjectedScreenshotExtractor

    @Inject
    lateinit var llmService: LLMService

    @Inject
    lateinit var accessibilityEventsRecorder: AccessibilityEventsRecorder

    private val serviceScope = CoroutineScope(Dispatchers.IO)

    override fun onCreate() {
        super.onCreate()
        startForeground(NOTIFICATION_ID, createNotification())
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        serviceScope.launch {
            handleIntent(intent)
        }
        return START_STICKY
    }

    private suspend fun handleIntent(intent: Intent?) {
        if (intent?.action == "com.sean8.jeminai.ACTION_TRIGGER_XRAY") {
            Log.d("ServerMockService", "Received trigger intent")
            captureAndSendData()
        }
    }

    private suspend fun captureAndSendData() {
        val accessibilityService = AppAccessibilityService.accessibilityComponent
        if (accessibilityService == null) {
            Log.e("ServerMockService", "Accessibility service not connected")
            return
        }

        // Capture XRAY
        val rootNode = accessibilityService.rootInActiveWindow
        if (rootNode == null) {
            Log.e("ServerMockService", "Failed to get root node")
            return
        }
        val treeRoot = TreeNode(rootNode)
        treeRoot.populateChildren()
        val xrayData = AccessibilityNodeInfoXmlDumper.getXml(treeRoot, accessibilityService.defaultDisplay)

        // Retrieve and reset accessibility events
        val events = accessibilityEventsRecorder.getAndResetEventHistory()
        val eventsData = events.joinToString(separator = "\n") { event ->
            val eventType = event.eventType.toString()
            val description = event.contentDescription?.toString() ?: ""
            val timestamp = event.eventTime
            "Event Type: $eventType, Description: $description, Timestamp: $timestamp"
        }

        // Capture Screenshot
        screenshotExtractor.start()
        delay(500) // Ensure the projection is started
        val bitmap = screenshotExtractor.acquireScreenshot()
        screenshotExtractor.stop()
        if (bitmap == null) {
            Log.e("ServerMockService", "Failed to capture screenshot")
            return
        }

        val screenshotByteArray = encodeBitmapToByteArray(bitmap)

        // Prepare the message
        val predefinedPrompt = "Your predefined prompt goes here."
        val message = """
            $predefinedPrompt

            XRAY Data:
            $xrayData

            Accessibility Events:
            $eventsData
        """.trimIndent()

        // Send message to LLM
        val llmResponse = llmService.sendMessage(message, screenshotByteArray)

        // Process the response
        processLLMResponse(llmResponse)
    }

    private fun encodeBitmapToByteArray(bitmap: Bitmap): ByteArray {
        val outputStream = ByteArrayOutputStream()
        bitmap.compress(Bitmap.CompressFormat.PNG, 100, outputStream)
        return outputStream.toByteArray()
    }

    private fun processLLMResponse(llmResponse: LLMResponse) {
        llmResponse.spoken?.let { spoken ->
            // Handle spoken response (e.g., text-to-speech)
            Log.d("ServerMockService", "Spoken: $spoken")
            // TODO: Implement TTS functionality
        }

        llmResponse.technical?.let { technical ->
            // Handle technical actions
            Log.d("ServerMockService", "Technical: $technical")
            // TODO: Implement action execution based on technical instructions
        }
    }

    override fun onBind(intent: Intent?): IBinder? {
        return null
    }

    private fun createNotification(): Notification {
        val notificationChannelId = "MEDIA_PROJECTION_CHANNEL"

        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val channel = NotificationChannel(
                notificationChannelId,
                "Media Projection Service",
                NotificationManager.IMPORTANCE_DEFAULT
            )
            val manager = getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
            manager.createNotificationChannel(channel)
        }

        val notificationBuilder = NotificationCompat.Builder(this, notificationChannelId)
            .setContentTitle("Screen Capture Active")
            .setContentText("Your screen is being captured")
            .setSmallIcon(android.R.drawable.ic_media_play)
            .setOngoing(true)

        return notificationBuilder.build()
    }

    companion object {
        private const val NOTIFICATION_ID = 1
    }
}
